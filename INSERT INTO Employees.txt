-- Query with LEAD() and LAG()
SELECT *,
       LAG(Salary) OVER (ORDER BY Salary) AS Previous_Salary,
       lead(salary)  OVER (ORDER BY Salary) AS next_Salary,
       FIRST_VALUE(Salary) OVER (ORDER BY Salary) AS Lowest_Salary,
       LAST_VALUE(Salary) OVER (ORDER BY Salary) AS Highest_Salary,
       NTH_VALUE(Salary, 2) OVER (ORDER BY Salary) AS Second_Lowest_Salary
FROM lead_lag;

SELECT 
    MONTH(orderdate) AS order_month,
     SUM(sales) AS total_sales,
    lag(sum(sales)) over(order by MONTH(orderdate)) as previous_sales
FROM 
    extract_monthfun
GROUP BY 
    MONTH(orderdate);

-- ORDER BY Clause

SELECT * 
FROM employeesdata
ORDER BY Salary DESC
LIMIT 5;

SELECT Name, Salary
FROM employeesdata
ORDER BY Name ASC;

SELECT DISTINCT Department
FROM employeesdata
ORDER BY Department;


-- GROUP BY

SELECT Department
FROM employeesdata
GROUP BY Department;

SELECT Department, COUNT(Department) AS total_count
FROM employeesdata
GROUP BY Department;

SELECT Gender, SUM(Salary) AS total_salary
FROM employeesdata
GROUP BY Gender
ORDER BY total_salary;

SELECT Gender, SUM(Salary) AS total_salary
FROM employeesdata
GROUP BY Gender
ORDER BY 1;


-- Practice Questions

-- 1) Total experience for each department
SELECT Department, SUM(Experience) AS total_experience
FROM employeesdata
GROUP BY Department;

-- 2) Maximum salary in each gender
SELECT Gender, MAX(Salary) AS max_salary
FROM employeesdata
GROUP BY Gender;

-- 3) Minimum age in each department
SELECT Department, MIN(Age) AS min_age
FROM employeesdata
GROUP BY Department;

-- 4) List of genders and number of employees earning above 60000
SELECT Gender, COUNT(*) AS total_employees
FROM employeesdata
WHERE Salary > 60000
GROUP BY Gender;


-- HAVING Clause

SELECT Department, AVG(Salary) AS avg_salary
FROM employeesdata
GROUP BY Department
HAVING AVG(Salary) > 60000;


-- Practice Questions

-- 1) Find department where the minimum salary is greater than 70000
SELECT Department, MIN(Salary) AS min_salary
FROM employeesdata
GROUP BY Department
HAVING MIN(Salary) > 70000;

-- 2) Find gender where the maximum salary is greater than 80000
SELECT Gender, MAX(Salary) AS max_salary
FROM employeesdata
GROUP BY Gender
HAVING MAX(Salary) > 80000;

-- 3) Find department where the average salary is greater than 60000
SELECT Department, AVG(Salary) AS avg_salary
FROM employeesdata
GROUP BY Department
HAVING AVG(Salary) > 60000;

-- 4) Find genders where the total salary is greater than 70000
SELECT Gender, SUM(Salary) AS total_salary
FROM employeesdata
GROUP BY Gender
HAVING SUM(Salary) > 70000;
